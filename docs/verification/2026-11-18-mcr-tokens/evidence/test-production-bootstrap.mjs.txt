import { createRequire } from 'node:module';
import { readFile } from 'node:fs/promises';
import { createServer } from 'node:http';
import path from 'node:path';
import assert from 'node:assert/strict';
const require=createRequire(path.join(process.cwd(),'package.json'));
const {chromium}=require('@playwright/test');
const root=path.join(process.cwd(),'apps/web/dist');

const server=createServer(async(req,res)=>{try{const name=decodeURIComponent(new URL(req.url,'http://localhost').pathname);const file=path.extname(name)?path.resolve(root,'.'+name):path.join(root,'index.html');if(!file.startsWith(root+path.sep)){res.writeHead(404);res.end();return;}const body=await readFile(file);if(name==='/theme-init.js')await new Promise(r=>setTimeout(r,200));res.writeHead(200,{'content-security-policy':`default-src 'self'; script-src 'self'; style-src 'self'; font-src 'self'`,'content-type':({'.js':'text/javascript','.html':'text/html','.css':'text/css','.woff2':'font/woff2','.png':'image/png'})[path.extname(file)]??'text/plain'});res.end(body);}catch{res.writeHead(404);res.end();}});
let browser;
try {
 await new Promise(r=>server.listen(0,'127.0.0.1',r));
 browser=await chromium.launch();const ctx=await browser.newContext();
 await ctx.addInitScript(()=>localStorage.setItem('mcr-theme','dark'));
 const page=await ctx.newPage();const requests=[];
 page.on('request',r=>{if(r.url().endsWith('/theme-init.js'))requests.push(r.url());});
 await page.route('**/assets/index-*.js',r=>r.fulfill({contentType:'text/javascript',body:'window.__themeAtMain = document.documentElement.dataset.theme;'}));
 await page.goto(`http://127.0.0.1:${server.address().port}/ingresar`,{waitUntil:'networkidle'});
 assert.equal(await page.locator('html').getAttribute('data-theme'),'dark','La preferencia debe aplicarse sin React');
 assert.equal(await page.evaluate(()=>window.__themeAtMain),'dark','El módulo principal debe comenzar con el tema resuelto');
 assert.equal(requests.length,1,'Un único controlador local, compatible con CSP self');
 assert.equal(await page.locator('script[src="/theme-init.js"]').evaluate(el=>el.defer),true);
 await page.evaluate(()=>{const script=document.createElement('script');script.textContent='window.__unexpectedInline = true';document.head.append(script);});
 assert.equal(await page.evaluate(()=>window.__unexpectedInline),undefined,'La CSP debe seguir bloqueando inline no autorizado');
 console.log(JSON.stringify({themeBeforeReact:'dark',bootstrapNetworkRequests:requests.length,unrelatedInlineBlocked:true}));
} finally {if(browser)await browser.close();await new Promise(r=>server.close(r));}
